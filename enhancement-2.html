Enhancement 02

Algorithms & Data Structures

This is the Travlr Getaways web application I built in CS-465 last semester. It uses MongoDB, Express, Angular, and Node.js. It has a customer website and an admin page where authenticated users can manage travel trip listings through an API.

I picked this project for the algorithms enhancement because the original trips endpoint just returned every record from the database with no way to search, sort, or page through results. I added server-side filtering by price range using MongoDB query operators, sorting by any field in ascending or descending order, and pagination with page and limit parameters. On the Angular side I added filter state properties and methods to the trip listing component and updated the data service to build query strings from filter parameters.

This enhancement covers course outcome 3 about designing and evaluating computing solutions using algorithmic principles and managing trade-offs in design choices.

const mongoose = require('mongoose');
const Trip = require('../models/travlr');
const Model = mongoose.model('trips');

// GET: /trips - lists all trips
// Supports: minPrice, maxPrice, sortBy, order, page, limit
const tripsList = async(req, res) => {
  try {
    let filter = {};

    if (req.query.minPrice || req.query.maxPrice) {
      filter.perPerson = {};
      if (req.query.minPrice) {
        filter.perPerson.$gte = Number(req.query.minPrice);
      }
      if (req.query.maxPrice) {
        filter.perPerson.$lte = Number(req.query.maxPrice);
      }
    }

    let sort = {};
    if (req.query.sortBy) {
      const order = req.query.order === 'desc' ? -1 : 1;
      sort[req.query.sortBy] = order;
    }

    const page = parseInt(req.query.page) || 1;
    const limit = parseInt(req.query.limit) || 0;
    const skip = (page - 1) * limit;

    let query = Model.find(filter);

    if (Object.keys(sort).length > 0) {
      query = query.sort(sort);
    }

    if (limit > 0) {
      query = query.skip(skip).limit(limit);
    }

    const q = await query.exec();

    if (!q || q.length === 0) {
      return res.status(404)
        .json({ message: 'No trips found' });
    }

    return res.status(200).json(q);
  } catch (err) {
    return res.status(500)
      .json({ message: 'Server error retrieving trips' });
  }
};

// GET: /trips/:tripCode
const tripsFindByCode = async(req, res) => {
  try {
    const q = await Model
      .find({'code' : req.params.tripCode})

trip-listing.component.ts 
import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { Router } from '@angular/router';

import { TripCardComponent } from '../trip-card/trip-card.component';
import { TripDataService } from '../services/trip-data.service';
import { Trip } from '../models/trip';
import { AuthenticationService } from '../services/authentication.service';

@Component({
  selector: 'app-trip-listing',
  standalone: true,
  imports: [CommonModule, TripCardComponent, FormsModule],
  templateUrl: './trip-listing.component.html',
  styleUrl: './trip-listing.component.css',
  providers: [TripDataService]
})
export class TripListingComponent implements OnInit {

  trips!: Trip[];
  message: string = '';

  // Filter and sort options
  minPrice: number | null = null;
  maxPrice: number | null = null;
  sortBy: string = '';
  sortOrder: string = 'asc';

  constructor(
    private tripDataService: TripDataService,
    private router: Router,
    private authenticationService: AuthenticationService
  ) {
    console.log('trip-listing constructor');
  }

  public addTrip(): void {
    this.router.navigate(['add-trip']);
  }

  public isLoggedIn(): boolean {
    return this.authenticationService.isLoggedIn();
  }

  public applyFilters(): void {
    this.getTrips();
  }

  public clearFilters(): void {
    this.minPrice = null;
    this.maxPrice = null;
    this.sortBy = '';
    this.sortOrder = 'asc';
    this.getTrips();
  }

  private getTrips(): void {
    let filters: any = {};

    if (this.minPrice) filters.minPrice = this.minPrice;
    if (this.maxPrice) filters.maxPrice = this.maxPrice;
    if (this.sortBy) {
      filters.sortBy = this.sortBy;
      filters.order = this.sortOrder;
    }

    this.tripDataService.getTrips(filters)
      .subscribe({
        next: (value: any) => {
          this.trips = value;
          if (value.length > 0) {
            this.message = 'There are ' + value.length
                + ' trips available.';
          } else {
            this.message = 'There were no trips retrieved'
                + ' from the database';
          }
          console.log(this.message);
        },
        error: (error: any) => {
          console.log('Error: ' + error);
        }
      });
  }

  ngOnInit(): void {
    console.log('ngOnInit');
    this.getTrips();
  }
}
